cmake_minimum_required(VERSION 3.20)
project(TCSModLoader LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(TCSModLoader SHARED src/loader.cpp src/native_pause_menu.cpp src/d3d9.def)
target_link_libraries(TCSModLoader PRIVATE user32 winmm)
set_target_properties(TCSModLoader PROPERTIES OUTPUT_NAME d3d9)
add_library(TCSFirstPersonPlugin SHARED src/mod.cpp)
target_link_libraries(TCSFirstPersonPlugin PRIVATE user32)
set_target_properties(TCSFirstPersonPlugin PROPERTIES OUTPUT_NAME TCSFirstPerson)
add_library(TCSModernOptionsPlugin SHARED src/modern_options.cpp)
set_target_properties(TCSModernOptionsPlugin PROPERTIES OUTPUT_NAME TCSModernOptions)
add_library(TCSSkipSplashPlugin SHARED src/skip_splash.cpp)
set_target_properties(TCSSkipSplashPlugin PROPERTIES OUTPUT_NAME TCSSkipSplash)
enable_testing()
add_executable(splash_patch_test tests/splash_patch_test.cpp)
target_include_directories(splash_patch_test PRIVATE src)
add_test(NAME splash_startup_patch COMMAND splash_patch_test)
add_executable(camera_math_test tests/camera_math_test.cpp)
target_include_directories(camera_math_test PRIVATE src)
add_test(NAME camera_math COMMAND camera_math_test)
add_executable(pause_menu_test tests/pause_menu_test.cpp)
target_include_directories(pause_menu_test PRIVATE src)
add_test(NAME pause_menu_dispatch COMMAND pause_menu_test)
add_executable(loader_menu_test tests/loader_menu_test.cpp src/native_pause_menu.cpp)
target_include_directories(loader_menu_test PRIVATE src)
target_link_libraries(loader_menu_test PRIVATE user32 winmm)
add_test(NAME per_mod_menu_navigation COMMAND loader_menu_test)
add_executable(mod_api_test tests/mod_api_test.cpp)
target_include_directories(mod_api_test PRIVATE src)
add_dependencies(mod_api_test TCSFirstPersonPlugin)
add_test(NAME mod_api_contract COMMAND mod_api_test $<TARGET_FILE:TCSFirstPersonPlugin>)
add_dependencies(mod_api_test TCSModernOptionsPlugin)
add_test(NAME modern_options_api_contract COMMAND mod_api_test $<TARGET_FILE:TCSModernOptionsPlugin> modern-options)

# Opt-in real-game integration probe, never shipped in the distribution ZIP.
option(TCS_BUILD_RUNTIME_PROBE "Build temporary native display lifecycle probe" OFF)
if(TCS_BUILD_RUNTIME_PROBE)
    add_library(TCSVideoRuntimeProbe SHARED tests/video_runtime_probe.cpp)
    target_include_directories(TCSVideoRuntimeProbe PRIVATE src)
endif()

add_library(TCSAllyDamagePlugin SHARED src/ally_damage.cpp)
set_target_properties(TCSAllyDamagePlugin PROPERTIES OUTPUT_NAME TCSAllyDamage)
add_executable(ally_damage_test tests/ally_damage_test.cpp)
target_include_directories(ally_damage_test PRIVATE src)
add_dependencies(ally_damage_test TCSAllyDamagePlugin)
add_test(NAME ally_damage_native_hooks COMMAND ally_damage_test $<TARGET_FILE:TCSAllyDamagePlugin>)
add_executable(ally_damage_runtime_test tests/ally_damage_runtime_test.cpp)
target_include_directories(ally_damage_runtime_test PRIVATE src)
target_link_options(ally_damage_runtime_test PRIVATE /BASE:0x10000000)
add_dependencies(ally_damage_runtime_test TCSAllyDamagePlugin)
add_test(NAME ally_damage_plugin_runtime COMMAND ally_damage_runtime_test $<TARGET_FILE:TCSAllyDamagePlugin>)

set_tests_properties(ally_damage_plugin_runtime PROPERTIES SKIP_RETURN_CODE 77)
add_executable(ally_combat_test tests/ally_combat_test.cpp)
target_include_directories(ally_combat_test PRIVATE src)
add_test(NAME ally_combat_decisions COMMAND ally_combat_test)
